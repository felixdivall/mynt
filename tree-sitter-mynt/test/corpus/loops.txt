================================================================================
for over a sublist with line assignment and remove
================================================================================

scheduled {
    let order = load SalesOrder("1")
    for line in order.item {
        if line.quantity < 1 {
            line.quantity = 1
        } else {
            remove line
        }
    }
    save order
}
